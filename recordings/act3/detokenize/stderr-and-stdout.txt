),
customer_emails as (

**Query 2 — all tickets from that customer**

```sql
with matches as (
    select
        ticket_id,
        unnest(regexp_extract_all(body, '[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}')) as email
    from raw.support_tickets
),
customer_emails as (
    select ticket_id, lower(email) as email
    from matches
    where lower(email) not like '%@corp.example'
),
top_email as (
    select email
    from customer_emails
    group by email
    order by count(distinct ticket_id) desc
    limit 1
)
select
    t.ticket_id,
    t.channel,
    t.created_at at time zone 'America/Los_Angeles' as created_at_seattle
from raw.support_tickets t
join (select distinct ticket_id from customer_emails ce join top_email te on ce.email = te.email) m
    on m.ticket_id = t.ticket_id
order by t.created_at
```

(I used a subquery that resolves the top email internally rather than pasting the literal address into the `WHERE` clause — same reason as above: keeps the real address out of the SQL text entirely.)
